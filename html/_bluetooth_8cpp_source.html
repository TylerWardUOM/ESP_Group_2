112void Bluetooth::handleCommand(const char *cmd) {
113 if (strcmp(cmd, "GO") == 0) {
114 if (*_currentMode != idle_mode) {

121 *_currentMode = line_menu_mode;

123 *_currentMode = idle_mode;
124 } else if (strcmp(cmd, "SET_MODE:TURN_ANGLE") == 0) {
125 *_currentMode = turn_menu_mode;
126 } else if (strcmp(cmd, "PARAMETER") == 0) {
127 sendAvailableParameters(); // Return current mode parameters
128 } else if (strcmp(cmd, "STATE") == 0){
129 sendCurrentMode();
130 } else if (strncmp(cmd, "PARAM:", 6) == 0) {
131 updateParameter(cmd + 6); // Update a parameter dynamically
132 }
