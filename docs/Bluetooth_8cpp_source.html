187// Parse incoming command
188void Bluetooth::parseCommand(char c) {
189 static char cmd_buffer[32];
190 static uint8_t cmd_index = 0;
191
192 if (c == '\n') {
193 cmd_buffer[cmd_index] = '\0'; // Null-terminate
194 handleCommand(cmd_buffer);
195 cmd_index = 0; // Reset
196 } else if (cmd_index < sizeof(cmd_buffer) - 1) {
197 cmd_buffer[cmd_index++] = c;
198 }
