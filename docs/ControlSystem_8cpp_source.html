150void ControlSystem::update() {
151 static bool timerStarted = false;
152 if (!timerStarted) {
153 pidTimer.start();
154 timerStarted = true;
155 }
156
157 float currentTime = pidTimer.read();
158 static float lastTime = currentTime;
159 float dt = currentTime - lastTime;
160 if (dt <= 0.0f) dt = 0.01f;
161 lastTime = currentTime;
162
163 switch (state) {
164 case MOVE_FORWARD:
165 processForwardMovement(dt);
166 break;
167 case TURNING:
168 processTurning(dt);
169 break;
170 case MOVING_SQUARE:
171 processSquare();
172 break;
173 case IDLE:
174 stopWheels();
175 pidTimer.stop();
176 pidTimer.reset();
177 timerStarted = false;
178 break;
179 case FOLLOW_LINE:
180 processFollowLine(dt);
181 break;
182 case BANG_BANG_PROPORTIONAL:
183 processBangBangProportional();
184 break;
185 case BANG_BANG:
186 processBangBang();
187 break;
188 }

309void ControlSystem::processSquare() {
310 switch (squareState) {
311 case MOVE_FORWARD_SQUARE:
312 // Handle square movement logic here
313 if (!moving) {
314 // Move forward one side of the square
315 moveForward(square_distance, square_speed);
316 moving = true;
317 }
318 if (isMovementComplete()) { // Wait for the movement to complete
319 wait(1);
320 // Determine whether to turn left or right
321 squareState = retracing ? TURN_RIGHT_SQUARE : TURN_LEFT_SQUARE;
322 moving = false;
323 }
324 break;
325
326 case TURN_LEFT_SQUARE:
327 // Turn left after moving forward (square pattern)
328 if (!moving) {
329 turn(90*square_left_turn_multiplier, square_speed); // Turn 90° left
330 moving = true;
331 }
332 if (isMovementComplete()) {
333 sideCount++; // Increment side count after a turn
334 moving = false;
335 squareState = (sideCount < 4) ? MOVE_FORWARD_SQUARE : TURN_AROUND_SQUARE;
336 }
337 break;
338
339 case TURN_RIGHT_SQUARE:
340 // Turn right while retracing
341 if (!moving) {
342 turn(-90*square_right_turn_multiplier, square_speed); // Turn 90° right (opposite direction for retracing)
343 moving = true;
344 }
345 if (isMovementComplete()) {
346 sideCount++; // Increment side count after a turn
347 moving = false;
348 squareState = (sideCount < 4) ? MOVE_FORWARD_SQUARE : STOP_SQUARE;
349 }
350 break;
351
352 case TURN_AROUND_SQUARE:
353 // Turn around (90) after completing the square
354 if (!moving) {
355 turn(90*square_left_turn_multiplier, square_speed); // Turn 180° to prepare for retracing
356 moving = true;
357 }
358 if (isMovementComplete()) {
359 retracing = true; // Start retracing
360 sideCount = 0; // Reset side count for retracing
361 squareState = MOVE_FORWARD_SQUARE; // Start moving forward again, but retracing
362 moving = false;
363 }
364 break;
365
366 case STOP_SQUARE:
367 // Stop after retracing all sides
368 if (isMovementComplete()){
369 retracing = false;
370 moving = false;
371 sideCount = 0;
372 squareCompleted = true;
373 state = IDLE;
374 squareState = IDLE_SQUARE;
375 stopWheels();
376 }
377 break;
378 }

442void ControlSystem::processBangBangProportional() {
443 float error = sensorArray.getError(); // Get error from sensor array
444
445 if (error>=900){
446 stopWheels();
447 movementCompleted = true;
448 state = IDLE;
449 }
450
451 if (fabs(error) > bangBangThreshold) {
452 // Use bang-bang for large errors
453 if (error > 0) {
454 leftWheel.setSpeed(basespeed);
455 rightWheel.setSpeed(basespeed * 0.5);
456 } else {
457 leftWheel.setSpeed(basespeed * 0.5);
458 rightWheel.setSpeed(basespeed);
459 }
460 } else {
461 // Use proportional control for small errors
462 float multiplier = 1.0f + kP * error;
463
464 // Clamp the multiplier
465 if (multiplier > maxMultiplier) multiplier = maxMultiplier;
466 if (multiplier < 1.0f / maxMultiplier) multiplier = 1.0f / maxMultiplier;
467
468 leftWheel.setSpeed(basespeed);
469 rightWheel.setSpeed(basespeed * multiplier);
