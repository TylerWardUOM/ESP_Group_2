133void ControlSystem::update() {
134 static bool timerStarted = false;
135 if (!timerStarted) {
136 pidTimer.start();
137 timerStarted = true;
138 }
139
140 float currentTime = pidTimer.read();
141 static float lastTime = currentTime;
142 float dt = currentTime - lastTime;
143 if (dt <= 0.0f) dt = 0.01f;
144 lastTime = currentTime;
145
146 switch (state) {
147 case MOVE_FORWARD:
148 processForwardMovement(dt);
149 break;
150 case TURNING:
151 processTurning(dt);
152 break;
153 case MOVING_SQUARE:
154 processSquare();
155 break;
156 case IDLE:
157 stopWheels();
158 pidTimer.stop();
159 pidTimer.reset();
160 timerStarted = false;
161 break;
162 case FOLLOW_LINE:
163 processFollowLine(dt);
164 break;
165 case BANG_BANG_PROPORTIONAL:
166 processBangBangProportional();
167 break;
168 case BANG_BANG:
169 processBangBang();
170 break;
171 }

290void ControlSystem::processSquare() {
291 switch (squareState) {
292 case MOVE_FORWARD_SQUARE:
293 // Handle square movement logic here
294 if (!moving) {
295 // Move forward one side of the square
296 moveForward(square_distance, square_speed);
297 moving = true;
298 }
299 if (isMovementComplete()) { // Wait for the movement to complete
300 wait(1);
301 // Determine whether to turn left or right
302 squareState = retracing ? TURN_RIGHT_SQUARE : TURN_LEFT_SQUARE;
303 moving = false;
304 }
305 break;
306
307 case TURN_LEFT_SQUARE:
308 // Turn left after moving forward (square pattern)
309 if (!moving) {
310 turn(90*square_left_turn_multiplier, square_speed); // Turn 90° left
311 moving = true;
312 }
313 if (isMovementComplete()) {
314 sideCount++; // Increment side count after a turn
315 moving = false;
316 squareState = (sideCount < 4) ? MOVE_FORWARD_SQUARE : TURN_AROUND_SQUARE;
317 }
318 break;
319
320 case TURN_RIGHT_SQUARE:
321 // Turn right while retracing
322 if (!moving) {
323 turn(-90*square_right_turn_multiplier, square_speed); // Turn 90° right (opposite direction for retracing)
324 moving = true;
325 }
326 if (isMovementComplete()) {
327 sideCount++; // Increment side count after a turn
328 moving = false;
329 squareState = (sideCount < 4) ? MOVE_FORWARD_SQUARE : STOP_SQUARE;
330 }
331 break;
332
333 case TURN_AROUND_SQUARE:
334 // Turn around (90) after completing the square
335 if (!moving) {
336 turn(90*square_left_turn_multiplier, square_speed); // Turn 180° to prepare for retracing
337 moving = true;
338 }
339 if (isMovementComplete()) {
340 retracing = true; // Start retracing
341 sideCount = 0; // Reset side count for retracing
342 squareState = MOVE_FORWARD_SQUARE; // Start moving forward again, but retracing
343 moving = false;
344 }
345 break;
346
347 case STOP_SQUARE:
348 // Stop after retracing all sides
349 if (isMovementComplete()){
350 retracing = false;
351 moving = false;
352 sideCount = 0;
353 squareCompleted = true;
354 state = IDLE;
355 squareState = IDLE_SQUARE;
356 stopWheels();
357 }
358 break;
359 }

426
427 if (error>=900){
428 stopWheels();

430 state = IDLE;

434 // Use bang-bang for large errors

436 leftWheel.setSpeed(basespeed);
437 rightWheel.setSpeed(basespeed * 0.5);
438 } else {
439 leftWheel.setSpeed(basespeed * 0.5);

441 }
442 } else {
443 // Use proportional control for small errors
444 float multiplier = 1.0f + kP * error;
445
446 // Clamp the multiplier
447 if (multiplier > maxMultiplier) multiplier = maxMultiplier;
448 if (multiplier < 1.0f / maxMultiplier) multiplier = 1.0f / maxMultiplier;
