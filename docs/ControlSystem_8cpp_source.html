220void ControlSystem::processSquare() {
221 switch (squareState) {
222 case MOVE_FORWARD_SQUARE:
223 // Handle square movement logic here
224 if (!moving) {
225 // Move forward one side of the square
226 moveForward(square_distance, square_speed);
227 moving = true;
228 }
229 if (isMovementComplete()) { // Wait for the movement to complete
230 wait(1);
231 // Determine whether to turn left or right
232 squareState = retracing ? TURN_RIGHT_SQUARE : TURN_LEFT_SQUARE;
233 moving = false;
234 }
235 break;
236
237 case TURN_LEFT_SQUARE:
238 // Turn left after moving forward (square pattern)
239 if (!moving) {
240 turn(90*square_left_turn_multiplier, square_speed); // Turn 90° left
241 moving = true;
242 }
243 if (isMovementComplete()) {
244 sideCount++; // Increment side count after a turn
245 moving = false;
246 squareState = (sideCount < 4) ? MOVE_FORWARD_SQUARE : TURN_AROUND_SQUARE;
247 }
248 break;
249
250 case TURN_RIGHT_SQUARE:
251 // Turn right while retracing
252 if (!moving) {
253 turn(-90*square_right_turn_multiplier, square_speed); // Turn 90° right (opposite direction for retracing)
254 moving = true;
255 }
256 if (isMovementComplete()) {
257 sideCount++; // Increment side count after a turn
258 moving = false;
259 squareState = (sideCount < 4) ? MOVE_FORWARD_SQUARE : STOP_SQUARE;
260 }
261 break;
262
263 case TURN_AROUND_SQUARE:
264 // Turn around (90) after completing the square
265 if (!moving) {
266 turn(90*square_left_turn_multiplier, square_speed); // Turn 180° to prepare for retracing
267 moving = true;
268 }
269 if (isMovementComplete()) {
270 retracing = true; // Start retracing
271 sideCount = 0; // Reset side count for retracing
272 squareState = MOVE_FORWARD_SQUARE; // Start moving forward again, but retracing
273 moving = false;
274 }
275 break;
276
277 case STOP_SQUARE:
278 // Stop after retracing all sides
279 if (isMovementComplete()){
280 retracing = false;
281 moving = false;
282 sideCount = 0;
283 squareCompleted = true;
284 state = IDLE;
285 squareState = IDLE_SQUARE;
286 stopWheels();
287 }
288 break;
289 }
290}
